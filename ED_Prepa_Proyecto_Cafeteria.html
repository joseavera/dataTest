Menu.cs — Módulo 1 C# 
using System;

class ModuloMenu
{
    // Arreglo de tamaño fijo = 6. El menú nunca cambia.
    // No usamos List porque no necesitamos que crezca.
    static string[] platillos = {
        "Tacos x2", "Torta de jamón", "Agua 500ml",
        "Jugo naranja", "Café americano", "Arroz con leche"
    };

    static double[] precios = {
        35.0, 45.0, 18.0,
        25.0, 22.0, 20.0
    };
    // ↑ Arreglos PARALELOS: platillos[i] ↔ precios[i]

    public static void MostrarMenu()
    {
        Console.WriteLine("=== MENÚ CAFETERÍA ===");
        for (int i = 0; i < platillos.Length; i++)
        {
            Console.WriteLine($"[{i}] {platillos[i],-20} ${precios[i]:F2}");
        }
    }

    public static void MostrarExtremos()
    {
        double max = precios[0], min = precios[0];
        int iMax = 0, iMin = 0;

        for (int i = 1; i < precios.Length; i++)
        {
            if (precios[i] > max) { max = precios[i]; iMax = i; }
            if (precios[i] < min) { min = precios[i]; iMin = i; }
        }

        Console.WriteLine($"Más caro:  {platillos[iMax]} (${max:F2})");
        Console.WriteLine($"Más barato: {platillos[iMin]} (${min:F2})");
    }
}

Alumnos.cs — Módulo 2 C# 
using System;
using System.Collections.Generic;

class ModuloAlumnos
{
    // LinkedList porque la lista CAMBIA constantemente.
    // Insertar/eliminar NO requiere mover otros elementos.
    static LinkedList<string> inscritos = new LinkedList<string>();

    public static void Inscribir(string nombre)
    {
        inscritos.AddLast(nombre); // O(1) — no mueve a nadie
        Console.WriteLine($"✔ {nombre} inscrito. Total: {inscritos.Count}");
    }

    public static void DarDeBaja(string nombre)
    {
        if (inscritos.Contains(nombre))
        {
            inscritos.Remove(nombre); // Solo redirige punteros
            Console.WriteLine($"✖ {nombre} dado de baja.");
        }
        else
            Console.WriteLine($"⚠ {nombre} no está inscrito.");
    }

    public static void MostrarInscritos()
    {
        Console.WriteLine("=== ALUMNOS INSCRITOS ===");
        foreach (string a in inscritos)
            Console.WriteLine($"  · {a}");
        Console.WriteLine($"Total: {inscritos.Count}");
    }
}

El módulo completo — las dos estructuras juntas código
Aquí está el módulo más complejo del sistema. Usa Queue<T> para la fila y Stack<T> para el historial. Léelo con atención.
Pedidos.cs — Módulo 3 C# 
using System;
using System.Collections.Generic;

class ModuloPedidos
{
    // FIFO: el primero en llegar es el primero en ser atendido
    static Queue<string> fila = new Queue<string>();

    // LIFO: para cancelar el último pedido registrado
    static Stack<string> historial = new Stack<string>();

    public static void RegistrarPedido(string alumno, string platillo)
    {
        string pedido = $"{alumno}: {platillo}";
        fila.Enqueue(pedido);      // entra al final de la fila
        historial.Push(pedido);    // se apila en el historial
        Console.WriteLine($"✔ Pedido registrado: {pedido}");
    }

    public static void AtenderSiguiente()
    {
        if (fila.Count == 0)
        {
            Console.WriteLine("⚠ No hay pedidos en espera.");
            return;
        }
        string siguiente = fila.Dequeue(); // sale del frente de la fila
        Console.WriteLine($"🍽 Atendiendo: {siguiente}");
    }

    public static void CancelarUltimo()
    {
        if (historial.Count == 0)
        {
            Console.WriteLine("⚠ No hay pedidos en el historial.");
            return;
        }
        // Pop saca el último pedido registrado
        string cancelado = historial.Pop();
        fila = new Queue<string>(historial); // reconstruye sin él
        Console.WriteLine($"✖ Cancelado: {cancelado}");
    }

    public static void VerEstado()
    {
        Console.WriteLine($"En fila: {fila.Count} · Historial: {historial.Count}");
        if (fila.Count > 0)
            Console.WriteLine($"  Siguiente a atender: {fila.Peek()}");
    }
}
